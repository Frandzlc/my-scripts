mainFrame.Parent = screenGui

-- حواف دائرية للإطار
local uiCorner = Instance.new("UICorner")

local titleCorner = Instance.new("UICorner")

-- كتابة معلومات البنق (Ping)
local pingLabel = Instance.new("TextLabel")

-- كتابة عدد اللاعبين
local playersLabel = Instance.new("TextLabel")

playersLabel.BackgroundTransparency = 1

playersLabel.Parent = mainFrame

-- زر تصغير/إغلاق خفيف أو توقيع إضافي باسمك في الأسفل
local footerLabel = Instance.new("TextLabel")
footerLabel.Size = UDim2.new(1, 0, 0, 20)
footerLabel.Position = UDim2.new(0, 0, 1, -20)
footerLabel.BackgroundTransparency = 1
footerLabel.TextColor3 = Color3.fromRGB(100, 100, 120)
footerLabel.TextSize = 11
footerLabel.Font = Enum.Font.Gotham
footerLabel.Text = "By r4w7"
footerLabel.Parent = mainFrame

-- حلقة تحديث البيانات تلقائياً كل ثانية
task.spawn(function()
	while task.wait(1) do
		-- حساب البنق بدقة من خصائص اللاعب
		local success, ping = pcall(function()
			return math.floor(player:GetNetworkPing() * 1000)
		end)
		
		if success and ping then
			pingLabel.Text = "📡 البنق: " .. ping .. " ms"
		else
			pingLabel.Text = "📡 البنق: غير متوفر"
		end
		
		-- حساب عدد اللاعبين الحاليين في السيرفر ضد الحد الأقصى
		local currentPlayers = #game.Players:GetPlayers()
		local maxPlayers = game.Players.MaxPlayers
		playersLabel.Text = "👥 اللاعبين: " .. currentPlayers .. " / " .. maxPlayers
	end
end)
