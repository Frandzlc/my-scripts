-- برمجة سحب الواجهة
local dragging, dragInput, dragStart, startPos
titleLabel.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging = true
		dragStart = input.Position
		startPos = mainFrame.Position
		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then
				dragging = false
			end
		end)
	end
end)

titleLabel.InputChanged:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
		dragInput = input
	end
end)

	if input == dragInput and dragging then
		local delta = input.Position - dragStart
		mainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
	end
end)

-- جدول الـ ESP
local espItems = {}
local function clearESP()
	for _, esp in pairs(espItems) do
		if esp then esp:Remove() end
	end
	espItems = {}
end

-- دالة الندرة
local function getRarity(name)
	local hash = 0
	for i = 1, #name do
		hash = hash + string.byte(name, i)
	end
	
	local rarityType = hash % 100
	if rarityType < 60 then
		return "عادي ⚪", Color3.fromRGB(200, 200, 200)
	elseif rarityType < 85 then
		return "نادر 🔵", Color3.fromRGB(50, 150, 255)
	elseif rarityType < 95 then
		return "أسطوري 🟣", Color3.fromRGB(180, 50, 255)
	else
		return "خرافي 🔥", Color3.fromRGB(255, 165, 0)
	end
end

-- دالة الانتقال الآمن (تمنع نظام الحماية من إرجاعك للخلف)
local function safeTeleport(targetPart)
	local character = player.Character
	if character and character:FindFirstChild("HumanoidRootPart") then
		local hrp = character.HumanoidRootPart
		-- إيقاف السرعة والحركة المؤقتة لمنع كشف الانتقال
		if character:FindFirstChildOfClass("Humanoid") then
			character:FindFirstChildOfClass("Humanoid").PlatformStand = true
		end
		
		-- الانتقال الفوري مع تثبيت الـ Velocity لتجاوز الـ Anti-Cheat
		hrp.CFrame = targetPart.CFrame + Vector3.new(0, 3, 0)
		hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
		
		task.wait(0.05)
		if character:FindFirstChildOfClass("Humanoid") then
			character:FindFirstChildOfClass("Humanoid").PlatformStand = false
		end
	end
end

-- التحديث المستمر
runService.RenderStepped:Connect(function()
	local character = player.Character
	if not character or not character:FindFirstChild("HumanoidRootPart") then
		clearESP()
		return
	end
	
	clearESP()
	
	local activeCount = 0
	if mainFrame.Visible then
		for _, child in pairs(scrollingFrame:GetChildren()) do
			if child:IsA("Frame") then child:Destroy() end
		end
	end
	
	for _, obj in pairs(workspace:GetDescendants()) do
		if obj ~= character and not obj:IsDescendantOf(character) then
			local humanoid = obj:IsA("Model") and obj:FindFirstChildOfClass("Humanoid")
			local isLive = obj:IsA("Model") and (humanoid or obj.Name:find("ضب") or obj.Name:find("Dhab") or obj.Name:find("Lizard"))
			
			if isLive then
				local part = obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart")
				if part then
					local dist = math.floor((character.HumanoidRootPart.Position - part.Position).Magnitude)
					
					if dist < 3000 then
						activeCount = activeCount + 1
						local rarityText, rarityColor = getRarity(obj.Name)
						
						-- رسم الـ ESP
						if espActive then
							local vector, onScreen = workspace.CurrentCamera:WorldToViewportPoint(part.Position)
							if onScreen then
								local espText = Drawing.new("Text")
								espText.Text = "🦎 " .. obj.Name .. " | " .. rarityText .. " [" .. dist .. "م]"
								espText.Size = 13
								espText.Center = true
								espText.Outline = true
								espText.Color = rarityColor
								espText.Position = Vector2.new(vector.X, vector.Y - 20)
								espText.Visible = true
								table.insert(espItems, espText)
							end
						end
						
						-- إضافة الضب للقائمة مع الانتقال الآمن عند الضغط عليه
						if mainFrame.Visible then
							local itemBtn = Instance.new("TextButton")
							itemBtn.Size = UDim2.new(1, -5, 0, 45)
							itemBtn.BackgroundColor3 = Color3.fromRGB(28, 28, 40)
							itemBtn.Text = ""
							itemBtn.Parent = scrollingFrame
							
							local itemCorner = Instance.new("UICorner")
							itemCorner.CornerRadius = UDim.new(0, 8)
							itemCorner.Parent = itemBtn
							
							local nameLabel = Instance.new("TextLabel")
							nameLabel.Size = UDim2.new(0.65, 0, 1, 0)
							nameLabel.Position = UDim2.new(0, 10, 0, 0)
							nameLabel.BackgroundTransparency = 1
							nameLabel.TextXAlignment = Enum.TextXAlignment.Left
							nameLabel.TextColor3 = rarityColor
							nameLabel.TextSize = 12
							nameLabel.Font = Enum.Font.GothamBold
							nameLabel.Text = obj.Name .. " (" .. rarityText .. ")"
							nameLabel.Parent = itemBtn
							
							local distLabel = Instance.new("TextLabel")
							distLabel.Size = UDim2.new(0.35, -10, 1, 0)
							distLabel.Position = UDim2.new(0.65, 0, 0, 0)
							distLabel.BackgroundTransparency = 1
							distLabel.TextXAlignment = Enum.TextXAlignment.Right
							distLabel.TextColor3 = Color3.fromRGB(0, 255, 200)
							distLabel.TextSize = 12
							distLabel.Font = Enum.Font.GothamMedium
							distLabel.Text = "انتقال ⚡ (" .. dist .. "م)"
							distLabel.Parent = itemBtn
							
							itemBtn.MouseButton1Click:Connect(function()
								safeTeleport(part)
							end)
						end
					end
				end
			end
		end
	end
	
	if mainFrame.Visible then
		scrollingFrame.CanvasSize = UDim2.new(0, 0, 0, activeCount * 51)
	end
end)
