-- دالة متقدمة للبحث عن الضبان أو الكائنات داخل الماب
local function getClosestDhab()
	local character = player.Character
	if not character or not character:FindFirstChild("HumanoidRootPart") then return nil end
	
	local closestTarget = nil
	local shortestDistance = math.huge
	
	-- نبحث في كل المجلدات عن أي نموذج أو جزء يحتوي على كاراكتر أو كائن متحرك أو اسم مشابه
	for _, obj in pairs(workspace:GetDescendants()) do
		if obj ~= character and not obj:IsDescendantOf(player.Character) then
			-- نتحقق إذا كان الكائن يحتوي على Humanoid (يعني كائن حي/موب) أو اسمه يهمنا
			local isLive = obj:IsA("Model") and obj:FindFirstChildOfClass("Humanoid")
			local name = string.lower(obj.Name)
			
			if isLive or name:find("ضب") or name:find("dhab") or name:find("lizard") or name:find("animal") or name:find("mob") then
				local part = nil
				if obj:IsA("Model") then
					part = obj.PrimaryPart or obj:FindFirstChild("HumanoidRootPart") or obj:FindFirstChildWhichIsA("BasePart")
				elseif obj:IsA("BasePart") then
					part = obj
				end
				
				if part then
					local distance = (character.HumanoidRootPart.Position - part.Position).Magnitude
					-- نتأكد أنه ليس اللاعب نفسه وأن المسافة معقولة
					if distance < shortestDistance and distance > 5 then
						shortestDistance = distance
						closestTarget = part
					end
				end
			end
		end
	end
	
	return closestTarget, math.floor(shortestDistance)
end

-- تحديث المسافة باستمرار
task.spawn(function()
	while task.wait(1) do
		local target, distance = getClosestDhab()
		if target then
			statusLabel.Text = "📍 أقرب ضب على بعد: " .." متر" .. distance
		else
			statusLabel.Text = "❌ لم يتم رصد ضبان قريبة"
		end
	end
end)

-- تنفيذ الانتقال عند الضغط على الزر
tpButton.MouseButton1Click:Connect(function()
	local target, _ = getClosestDhab()
	local character = player.Character
	
	if target and character and character:FindFirstChild("HumanoidRootPart") then
		character.HumanoidRootPart.CFrame = target.CFrame + Vector3.new(0, 3, 0)
		statusLabel.Text = "⚡ تم الانتقال بنجاح!"
	else
		statusLabel.Text = "⚠️ تعذر العثور على هدف للانتقال!"
	end
end)
