-- دالة لفتح وإغلاق القائمة عبر زر ESC
userInputService.InputBegan:Connect(function(input, gameProcessed)
	if input.KeyCode == Enum.KeyCode.Escape then
		-- منع اللعبة من إيقاف القائمة الافتراضية بشكل كامل أو السماح بفتحها
		mainFrame.Visible = not mainFrame.Visible
	end
end)

-- حلقة لتحديث رادار الضبان والمسافات والأنواع باستمرار
task.spawn(function()
	while task.wait(0.5) do
		if mainFrame.Visible then
			local character = player.Character
			if character and character:FindFirstChild("HumanoidRootPart") then
				-- مسح القائمة القديمة
				for _, child in pairs(scrollingFrame:GetChildren()) do
					if child:IsA("Frame") then
						child:Destroy()
					end
				end
				
				local count = 0
				-- البحث عن جميع الكائنات المحتملة في الـ Workspace
				for _, obj in pairs(workspace:GetDescendants()) do
					if obj ~= character and not obj:IsDescendantOf(character) then
						local isLive = obj:IsA("Model") and (obj:FindFirstChildOfClass("Humanoid") or obj.Name:find("ضب") or obj.Name:find("Dhab") or obj.Name:find("Lizard"))
						
						if isLive then
							local part = obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart")
							if part then
								local dist = math.floor((character.HumanoidRootPart.Position - part.Position).Magnitude)
								
								if dist < 1500 then -- نطاق البحث 1500 متر
									count = count + 1
									
									-- تصميم خانة لكل ضب يتم رصده
									local itemFrame = Instance.new("Frame")
									itemFrame.Size = UDim2.new(1, -5, 0, 45)
									itemFrame.BackgroundColor3 = Color3.fromRGB(30, 30, 45)
									itemFrame.Parent = scrollingFrame
									
									local itemCorner = Instance.new("UICorner")
									itemCorner.CornerRadius = UDim.new(0, 8)
									itemCorner.Parent = itemFrame
									
									-- اسم ونوع الضب
									local nameLabel = Instance.new("TextLabel")
									nameLabel.Size = UDim2.new(0.6, 0, 1, 0)
									nameLabel.Position = UDim2.new(0, 10, 0, 0)
									nameLabel.BackgroundTransparency = 1
									nameLabel.TextXAlignment = Enum.TextXAlignment.Left
									nameLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
									nameLabel.TextSize = 13
									nameLabel.Font = Enum.Font.GothamBold
									nameLabel.Text = "🦎 " .. obj.Name
									nameLabel.Parent = itemFrame
									
									-- المسافة عنه
									local distLabel = Instance.new("TextLabel")
									distLabel.Size = UDim2.new(0.4, -10, 1, 0)
									distLabel.Position = UDim2.new(0.6, 0, 0, 0)
									distLabel.BackgroundTransparency = 1
									distLabel.TextXAlignment = Enum.TextXAlignment.Right
									distLabel.TextColor3 = Color3.fromRGB(0, 255, 200)
									distLabel.TextSize = 13
									distLabel.Font = Enum.Font.GothamMedium
									distLabel.Text = dist .. " متر"
									distLabel.Parent = itemFrame
								end
							end
						end
					end
				end
				
				-- تحديث حجم القائمة بناءً على عدد النتائج
				scrollingFrame.CanvasSize = UDim2.new(0, 0, 0, count * 51)
			end
		end
	end
end)
