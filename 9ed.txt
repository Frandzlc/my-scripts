-- دالة انتقال فحص شامل لجميع مجلدات الماب
btn.MouseButton1Click:Connect(function()
	local char = player.Character
	if not char or not char:FindFirstChild("HumanoidRootPart") then return end
	
	local found = false
	status.Text = "🔍 جاري المسح الشامل..."
	
	-- نبحث عن أي كائن في الـ Workspace يملك Humanoid أو Pivot أو جزء مجسم
	for _, item in pairs(workspace:GetChildren()) do
		if item ~= char and item.Name ~= "Terrain" and item.Name ~= "Camera" then
			-- فحص المجلدات والمجسمات
			local targetPart = nil
			
			if item:IsA("Model") then
				targetPart = item.PrimaryPart or item:FindFirstChildWhichIsA("BasePart")
			elseif item:IsA("BasePart") and not item.Anchored then
				targetPart = item
			end
			
			-- نضمن أنه ليس لاعباً آخر
			if targetPart and not game.Players:GetPlayerFromCharacter(item) then
				local dist = (char.HumanoidRootPart.Position - targetPart.Position).Magnitude
				if dist > 8 and dist < 2000 then
					char.HumanoidRootPart.CFrame = targetPart.CFrame + Vector3.new(0, 4, 0)
					status.Text = "✅ تم الانتقال إلى: " .. item.Name
					found = true
					break
				end
			end
		end
	end
	
	if not found then
		status.Text = "⚠️ اخرج للصخراء واضغط الزر مجدداً"
	end
end)
