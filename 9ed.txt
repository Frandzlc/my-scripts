-- متغير لتفعيل أو تعطيل القتل الفوري
local instantKillActive = false

-- زر تفعيل القتل بطلقة واحدة (Instant Kill Toggle)
local killToggleBtn = Instance.new("TextButton")
killToggleBtn.Size = UDim2.new(1, -20, 0, 35)
killToggleBtn.Position = UDim2.new(0, 10, 0, 48)
killToggleBtn.BackgroundColor3 = Color3.fromRGB(60, 30, 30)
killToggleBtn.Text = "⚡ قتل بطلقة واحدة: [ معطل ❌ ]"
killToggleBtn.TextColor3 = Color3.fromRGB(255, 100, 100)
killToggleBtn.TextSize = 13
killToggleBtn.Font = Enum.Font.GothamBold
killToggleBtn.Parent = mainFrame

local toggleCorner = Instance.new("UICorner")
toggleCorner.CornerRadius = UDim.new(0, 8)
toggleCorner.Parent = killToggleBtn

killToggleBtn.MouseButton1Click:Connect(function()
	instantKillActive = not instantKillActive
	if instantKillActive then
		killToggleBtn.BackgroundColor3 = Color3.fromRGB(30, 70, 40)
		killToggleBtn.Text = "⚡ قتل بطلقة واحدة: [ مفعل ✅ ]"
		killToggleBtn.TextColor3 = Color3.fromRGB(100, 255, 150)
	else
		killToggleBtn.BackgroundColor3 = Color3.fromRGB(60, 30, 30)
		killToggleBtn.Text = "⚡ قتل بطلقة واحدة: [ معطل ❌ ]"
		killToggleBtn.TextColor3 = Color3.fromRGB(255, 100, 100)
	end
end)

-- فتح وإغلاق القائمة بزر ESC
userInputService.InputBegan:Connect(function(input, gameProcessed)
	if input.KeyCode == Enum.KeyCode.Escape then
		mainFrame.Visible = not mainFrame.Visible
	end
end)

-- جدول تخزين الـ ESP
local espItems = {}

local function clearESP()
	for _, esp in pairs(espItems) do
		if esp then
			esp:Remove()
		end
	end
	espItems = {}
end

-- التحديث المستمر للـ ESP والقتل الفوري والرادار
runService.RenderStepped:Connect(function()
	local character = player.Character
	if not character or not character:FindFirstChild("HumanoidRootPart") then
		clearESP()
		return
	end
	
	clearESP()
	
	local activeCount = 0
	if mainFrame.Visible then
		for _, child in pairs(scrollingFrame:GetChildren()) do
			if child:IsA("Frame") then
				child:Destroy()
			end
		end
	end
	
	for _, obj in pairs(workspace:GetDescendants()) do
		if obj ~= character and not obj:IsDescendantOf(character) then
			local humanoid = obj:IsA("Model") and obj:FindFirstChildOfClass("Humanoid")
			local isLive = obj:IsA("Model") and (humanoid or obj.Name:find("ضب") or obj.Name:find("Dhab") or obj.Name:find("Lizard"))
			
			if isLive then
				local part = obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart")
				if part then
					local dist = math.floor((character.HumanoidRootPart.Position - part.Position).Magnitude)
					
					if dist < 2000 then
						-- تنفيذ القتل الفوري إذا كان الزر مفعل ولديه Humanoid
						if instantKillActive and humanoid and humanoid.Health > 0 then
							humanoid.Health = 0
						end
						
						activeCount = activeCount + 1
						
						-- رسم الـ ESP
						local vector, onScreen = workspace.CurrentCamera:WorldToViewportPoint(part.Position)
						if onScreen then
							local espText = Drawing.new("Text")
							espText.Text = "🦎 " .. obj.Name .. " [" .. dist .. "م]"
							espText.Size = 14
							espText.Center = true
							espText.Outline = true
							espText.Color = instantKillActive and Color3.fromRGB(255, 50, 50) or Color3.fromRGB(0, 255, 150)
							espText.Position = Vector2.new(vector.X, vector.Y - 20)
							espText.Visible = true
							table.insert(espItems, espText)
						end
						
						-- إضافة القائمة
						if mainFrame.Visible then
							local itemFrame = Instance.new("Frame")
							itemFrame.Size = UDim2.new(1, -5, 0, 45)
							itemFrame.BackgroundColor3 = Color3.fromRGB(30, 30, 45)
							itemFrame.Parent = scrollingFrame
							
							local itemCorner = Instance.new("UICorner")
							itemCorner.CornerRadius = UDim.new(0, 8)
							itemCorner.Parent = itemFrame
							
							local nameLabel = Instance.new("TextLabel")
							nameLabel.Size = UDim2.new(0.6, 0, 1, 0)
							nameLabel.Position = UDim2.new(0, 10, 0, 0)
							nameLabel.BackgroundTransparency = 1
							nameLabel.TextXAlignment = Enum.TextXAlignment.Left
							nameLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
							nameLabel.TextSize = 13
							nameLabel.Font = Enum.Font.GothamBold
							nameLabel.Text = "🦎 " .. obj.Name
							nameLabel.Parent = itemFrame
							
							local distLabel = Instance.new("TextLabel")
							distLabel.Size = UDim2.new(0.4, -10, 1, 0)
							distLabel.Position = UDim2.new(0.6, 0, 0, 0)
							distLabel.BackgroundTransparency = 1
							distLabel.TextXAlignment = Enum.TextXAlignment.Right
							distLabel.TextColor3 = Color3.fromRGB(0, 255, 200)
							distLabel.TextSize = 13
							distLabel.Font = Enum.Font.GothamMedium
							distLabel.Text = dist .. " متر"
							distLabel.Parent = itemFrame
						end
					end
				end
			end
		end
	end
	
	if mainFrame.Visible then
		scrollingFrame.CanvasSize = UDim2.new(0, 0, 0, activeCount * 51)
	end
end)
