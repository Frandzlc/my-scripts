-- جدول لتخزين عناصر الـ ESP لكل ضب
local espItems = {}

-- دالة لإزالة الـ ESP القديم
local function clearESP()
	for _, esp in pairs(espItems) do
		if esp then
			esp:Remove()
		end
	end
	espItems = {}
end
